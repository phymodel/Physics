# 时间图像

## 知识点

$t$图像

## 时间图像

- 时刻表述：“第 $1\,\rm{s}$ 初”、“第 $1\,\rm{s}$ 末、第 $2\,\rm{s}$ 初”、“前 $4 \,\rm{s}$ 末”。
- 时间间隔表述：“第 $1\,\rm{s}$ 内，前 $1\,\rm{s}$”、“第 $2\,\rm{s}$ 内”、“前 $2\,\rm{s}$”、“第二个 $2\,\rm{s}$ 内”、“第 $5\,\rm{s}$ 内”。

## 设计交互场景

- 实体
- 实体_时刻表述
- 触发系统_时刻表述
- 实体_时间间隔表述
- 触发系统_时间间隔表述
- 实体_场景切换
- 触发系统_场景切换

### 实体

```phymodel

Latex()
.name("textConcept")
.mesh(width: 0.6, textPath: "时间图像#时间图像")
.material(red: 1.0, green: 1.0, blue: 1.0, opacity: 0.999, roughness: 0.5, metallic: 0.8)
.physic(mass: 0.1, staticFriction: 0.9, dynamicFriction: 0.75, restitution: 0.22, mode: .kinematic)
.position(x: 0.0, y: 1.8, z: 0.0)

// 打组系统
Group(name: "CoordinateSystem", position: (0.0, 1.5, 0.0), gesture: false) {
    
    CAD()
    .name("CoordinateSystemBodyArrow")
    .mesh(commands: [
        "DEF arrow = PATH_OPEN 0,0,0 Z 2.9,0 2.8,-0.03 2.8,-0.01 0,-0.01 0,0  REV X 360"
    ])
    .material(red: 0.5, green: 0.5, blue: 0.5, opacity: 1, roughness: 0.3, metallic: 0.9)
    .physic(mass: 0.5, staticFriction: 0.6, dynamicFriction: 0.4, restitution: 0.2, mode: .kinematic)
    .position(x: -1.4, y: 0.0, z: 0.0)
    .gesture(false)
    
    Text()
    .name("unitLength")
    .mesh(text: "t", extrusionDepth: 0.001, font: "Times New Roman Italic", fontSize: 0.05)
    .material(red: 0.9, green: 0.1, blue: 0.3, opacity: 1, roughness: 0.5, metallic: 0.8)
    .physic(mass: 0.3, staticFriction: 0.6, dynamicFriction: 0.4, restitution: 0.2, mode: .kinematic)
    .position(x: 1.4, y: -0.08, z: 0.0)
    .gesture(false)
    
    Text()
    .name("unit")
    .mesh(text: "/s", extrusionDepth: 0.001, font: "Times New Roman", fontSize: 0.05)
    .material(red: 0.9, green: 0.1, blue: 0.3, opacity: 1, roughness: 0.5, metallic: 0.8)
    .physic(mass: 0.3, staticFriction: 0.6, dynamicFriction: 0.4, restitution: 0.2, mode: .kinematic)
    .position(x: 1.43, y: -0.08, z: 0.0)
    .gesture(false)
    
    // 批量生成元素（刻度）
    Repeat(7) { i in
        CAD()
        .name("CoordinateSystemEllipsoid[i]")
        .mesh(commands: [
            "DEF sphere = SOLID_FORMULA 0,0,0 (x/0.02)^2+(y/0.02)^2+(z/0.02)^2-1=0",
        ])
        .physic(mass: 0.6, staticFriction: 0.6, dynamicFriction: 0.4, restitution: 0.2, mode: .kinematic)
        .material(red: 0.67, green: 0.32, blue: 1.0, opacity: 1, roughness: 0.5, metallic: 0.8)
        .position(x: -1.2 + i * 0.4, y: 0.0, z: 0.0)
        .gesture(false)
    }
    
    Repeat(7) { i in
        Text()
        .name("CoordinateSystemText[i]")
        .mesh(text: "[i]", extrusionDepth: 0.001, font: "Times New Roman", fontSize: 0.05)
        .material(red: 0.9, green: 0.1, blue: 0.3, opacity: 1, roughness: 0.5, metallic: 0.8)
        .physic(mass: 0.3, staticFriction: 0.6, dynamicFriction: 0.4, restitution: 0.2, mode: .kinematic)
        .position(x: -1.2 + i * 0.4, y: -0.08, z: 0.0)
        .gesture(false)
    }
    
}
```
